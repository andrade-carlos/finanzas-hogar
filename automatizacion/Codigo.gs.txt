function lfRpc_(c, fn, args) {
  var r = UrlFetchApp.fetch(c.u + '/rest/v1/rpc/' + fn, { method: 'post', contentType: 'application/json', headers: { apikey: c.k }, payload: JSON.stringify(args), muteHttpExceptions: true });

  if (code >= 300) {

nacl.setPRNG(function (x, n) {
  var i = 0;
  while (i < n) {
    var d = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, Utilities.getUuid() + Utilities.getUuid() + (++LF_CTR) + ':' + Date.now());
    for (var j = 0; j < d.length && i < n; j++) x[i++] = d[j] & 255;
  }
});
function lfU8_(bytes) { var u = new Uint8Array(bytes.length); for (var i = 0; i < bytes.length; i++) u[i] = bytes[i] & 255; return u; }
function lfSigned_(u8) { var a = new Array(u8.length); for (var i = 0; i < u8.length; i++) a[i] = u8[i] > 127 ? u8[i] - 256 : u8[i]; return a; }
function lfSeal_(obj, pkB64) {
  var plain = lfU8_(Utilities.gzip(Utilities.newBlob(JSON.stringify(obj), 'application/json')).getBytes());
  var eph = nacl.box.keyPair(), nonce = nacl.randomBytes(24);
  var box = nacl.box(plain, nonce, lfU8_(Utilities.base64Decode(pkB64)), eph.secretKey);
  var out = new Uint8Array(60 + box.length);
  out.set(LF_MAGIC, 0); out.set(eph.publicKey, 4); out.set(nonce, 36); out.set(box, 60);
  return Utilities.base64Encode(lfSigned_(out));
}
function lfOpenSummary_(ctB64, keyB64) {
  if (!ctB64 || !keyB64) return null;
  var all = lfU8_(Utilities.base64Decode(ctB64));
  var pt = nacl.secretbox.open(all.subarray(24), all.subarray(0, 24), lfU8_(Utilities.base64Decode(keyB64)));
  return pt ? JSON.parse(Utilities.newBlob(lfSigned_(pt)).getDataAsString()) : null;
}

// ---------------------------------------------------------------- instalación
function instalar() {
  var c = lfCfg_();
  var st = lfRpc_(c, 'ingest_status', { p_token: c.t });
  if (st.pk !== c.p) throw new Error('Las claves de la app cambiaron. Genera un código nuevo en la app.');
  if (CODIGO_CONFIGURACION) LF_PROPS.setProperty('LF_CFG', CODIGO_CONFIGURACION.trim());
  if (!LF_PROPS.getProperty('LF_WM')) LF_PROPS.setProperty('LF_WM', String(Date.parse((c.d || new Date(Date.now() - 25 * 864e5).toISOString().slice(0, 10)) + 'T00:00:00-05:00')));
  ScriptApp.getProjectTriggers().forEach(function (t) { ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('procesar').timeBased().everyMinutes(15).create();
  ScriptApp.newTrigger('resumenSemanal').timeBased().onWeekDay(ScriptApp.WeekDay.MONDAY).atHour(7).nearMinute(13).create();
  Logger.log('Conectado como "' + st.label + '". Leyendo correos desde ' + new Date(+LF_PROPS.getProperty('LF_WM')).toLocaleDateString() + '…');
  var r = procesar();
  Logger.log('Listo. Enviados: ' + r.sent + ', omitidos: ' + r.skipped + (r.more ? ' (quedan más; se enviarán en los próximos minutos)' : '') + '.');
  Logger.log('Ya puedes borrar el código de la línea CODIGO_CONFIGURACION (quedó guardado de forma privada en el proyecto).');
}
function desinstalar() {
  ScriptApp.getProjectTriggers().forEach(function (t) { ScriptApp.deleteTrigger(t); });
  LF_PROPS.deleteAllProperties();
  Logger.log('Programa detenido y configuración borrada.');
}

// ---------------------------------------------------------------- envío de avisos
function procesar() {
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(5000)) return { sent: 0, skipped: 0 };
  var t0 = Date.now(), c, res = { sent: 0, skipped: 0, more: false };
  try {
    c = lfCfg_();
    var wm = +LF_PROPS.getProperty('LF_WM') || (Date.now() - 3 * 864e5);
    var sent = JSON.parse(LF_PROPS.getProperty('LF_SENT') || '[]');
    var q = MailParse.gmailQuery(REMITENTES_EXTRA) + ' after:' + Math.floor((wm - 3 * 3600e3) / 1000);
    var msgs = [];
    for (var page = 0; page < 5; page++) {
      var th = GmailApp.search(q, page * 100, 100);
      th.forEach(function (t) { t.getMessages().forEach(function (m) { if (m.getDate().getTime() >= wm - 3 * 3600e3) msgs.push(m); }); });
      if (th.length < 100) break;
    }
    msgs.sort(function (a, b) { return a.getDate() - b.getDate(); });
    var batch = [], batchIds = [], lastDate = wm;
    var flush = function () {
      if (!batch.length) return;
      lfRpc_(c, 'ingest_push', { p_token: c.t, p_items: batch });
      batchIds.forEach(function (x) { sent.push(x.slice(0, 16)); });
      res.sent += batch.length; batch = []; batchIds = [];
    };
    for (var i = 0; i < msgs.length; i++) {
      if (Date.now() - t0 > 270000) { res.more = true; break; }
      var m = msgs[i], ext = lfHex_(c.n + ':' + m.getId());
      if (sent.indexOf(ext.slice(0, 16)) >= 0) { lastDate = Math.max(lastDate, m.getDate().getTime()); continue; }
      var body = m.getPlainBody() || '';
      var atts = m.getAttachments({ includeInlineImages: false }).filter(function (a) { return /\.pdf$/i.test(a.getName()) || /pdf/i.test(a.getContentType()); });
      var payload = { v: 1, mid: m.getId(), label: c.n, from: m.getFrom(), subject: m.getSubject(), date: m.getDate().toISOString(), body: body.slice(0, 15000), att: [] };
      var ev = MailParse.parse({ from: payload.from, subject: payload.subject, date: payload.date, body: body, attachments: atts.map(function (a) { return a.getName(); }) });
      if (ev.type === 'ignore' || ev.type === 'duplicate_notice' || (ev.type === 'statement' && !atts.length)) { res.skipped++; sent.push(ext.slice(0, 16)); lastDate = Math.max(lastDate, m.getDate().getTime()); continue; }
      if (ev.type === 'statement') {
        flush();
        var pdf = atts[0], bytes = pdf.getBytes();
        if (bytes.length > 4000000) { res.skipped++; sent.push(ext.slice(0, 16)); continue; }
        payload.att = [{ name: pdf.getName(), b64: Utilities.base64Encode(bytes) }];
        lfRpc_(c, 'ingest_push', { p_token: c.t, p_items: [{ ext: ext, kind: 'pdf', ct: lfSeal_(payload, c.p) }] });
        sent.push(ext.slice(0, 16)); res.sent++;
      } else {
        payload.parsed = ev;
        batch.push({ ext: ext, kind: 'mail', ct: lfSeal_(payload, c.p) }); batchIds.push(ext);
        if (batch.length >= 20) flush();
      }
      lastDate = Math.max(lastDate, m.getDate().getTime());
    }
    flush();
    if (sent.length > 600) sent = sent.slice(-600);
    LF_PROPS.setProperty('LF_SENT', JSON.stringify(sent));
    if (!res.more) LF_PROPS.setProperty('LF_WM', String(Math.max(wm, lastDate - 3600e3)));
    else LF_PROPS.setProperty('LF_WM', String(Math.max(wm, lastDate)));
    LF_PROPS.deleteProperty('LF_FAILS');
    if (res.sent === 0) lfRpc_(c, 'ping', {});
  } catch (e) {
    lfFail_(c, e);
    throw e;
  } finally { lock.releaseLock(); }
  return res;
}
function lfFail_(c, e) {
  var n = +(LF_PROPS.getProperty('LF_FAILS') || 0) + 1;
  LF_PROPS.setProperty('LF_FAILS', String(n));
  try { if (c) lfRpc_(c, 'ingest_report', { p_token: c.t, p_error: String(e.message || e).slice(0, 280) }); } catch (x) { }
  var last = +(LF_PROPS.getProperty('LF_FAIL_MAIL') || 0);
  if (n >= 3 && Date.now() - last > 864e5) {
    LF_PROPS.setProperty('LF_FAIL_MAIL', String(Date.now()));
    MailApp.sendEmail(Session.getEffectiveUser().getEmail(), 'Finanzas del hogar: la automatización del correo tiene un problema',
      'El programa que envía tus avisos bancarios a la app falló ' + n + ' veces seguidas.\n\nMotivo: ' + (e.message || e) + '\n\nNo se perdió nada: cuando se corrija, enviará lo pendiente. Abre script.google.com › tu proyecto › Ejecuciones para ver el detalle.');
  }
}

// ---------------------------------------------------------------- resumen semanal
function resumenSemanal() {
  var c = lfCfg_();
  var st = lfRpc_(c, 'ingest_status', { p_token: c.t });
  var snap = null; try { snap = lfOpenSummary_(st.summary, c.s); } catch (e) { }
  var since = new Date(Date.now() - 7 * 864e5);
  var th = GmailApp.search(MailParse.gmailQuery(REMITENTES_EXTRA) + ' after:' + Math.floor(since.getTime() / 1000), 0, 200), evs = [];
  th.forEach(function (t) { t.getMessages().forEach(function (m) { if (m.getDate() < since) return; var ev = MailParse.parse({ from: m.getFrom(), subject: m.getSubject(), date: m.getDate().toISOString(), body: m.getPlainBody() || '', attachments: [] }); if (ev.type !== 'ignore' && ev.type !== 'duplicate_notice' && ev.type !== 'statement') evs.push(ev); }); });
  evs.sort(function (a, b) { return a.date < b.date ? -1 : 1; });
  var html = lfSummaryHtml_(snap, st, evs, c);
  var to = ENVIAR_RESUMEN_A || Session.getEffectiveUser().getEmail();
  MailApp.sendEmail({ to: to, subject: 'Resumen semanal de finanzas · ' + Utilities.formatDate(new Date(), 'America/Guayaquil', 'd MMM yyyy'), htmlBody: html, name: 'Finanzas del hogar' });
}
function lfEsc_(s) { return String(s === null || s === undefined ? '' : s).replace(/[&<>"]/g, function (x) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[x]; }); }
function lfMoney_(x) { return '$' + (Math.round((+x || 0) * 100) / 100).toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 }); }
function lfSummaryHtml_(s, st, evs, c) {
  var TYPE = { card_purchase: 'Consumo tarjeta', debit_purchase: 'Compra débito', transfer_out: 'Transferencia enviada', transfer_in: 'Transferencia recibida', own_transfer: 'Entre tus cuentas', card_payment: 'Pago de tarjeta', card_payment_received: 'Pago recibido', loan_payment: 'Pago de préstamo', bill: 'Planilla', bg_transfer: 'Transferencia', savings_in: 'Aporte ahorro', savings_out: 'Retiro ahorro', savings_interest: 'Intereses ahorro', unknown: 'Otro' };
  var td = 'style="padding:6px 8px;border-bottom:1px solid #e5e9f0"', h = '<div style="font-family:Arial,sans-serif;color:#142033;max-width:640px">';
  h += '<h2 style="margin:0 0 4px">Resumen semanal de finanzas</h2><p style="color:#667;margin:0 0 16px">' + lfEsc_(c.n) + '</p>';
  if (s) {
    h += '<h3>' + lfEsc_(s.monthLabel) + '</h3><table style="border-collapse:collapse;width:100%">' +
      '<tr><td ' + td + '>Ingresos del mes' + (s.realIncome ? '' : ' (sueldo aún estimado)') + '</td><td ' + td + ' align="right"><b>' + lfMoney_(s.income) + '</b></td></tr>' +
      '<tr><td ' + td + '>Gastos fijos y planillas presupuestados</td><td ' + td + ' align="right">' + lfMoney_(s.bills) + '</td></tr>' +
      '<tr><td ' + td + '>Deuda total</td><td ' + td + ' align="right"><b>' + lfMoney_(s.debt.total) + '</b>' + (s.debt.baseline ? '<br><span style="color:#2f855a;font-size:12px">' + Math.round((1 - s.debt.total / s.debt.baseline) * 100) + '% eliminado desde el inicio</span>' : '') + '</td></tr>' +
      (s.debt.freeMonth ? '<tr><td ' + td + '>Libres de deudas (según el plan)</td><td ' + td + ' align="right">' + lfEsc_(s.debt.freeMonth) + '</td></tr>' : '') +
      '<tr><td ' + td + '>Fondo de emergencia</td><td ' + td + ' align="right">' + lfMoney_(s.emergency) + '</td></tr></table>';
    if (s.spent && s.spent.length) h += '<h3>Gasto del mes por categoría</h3><table style="border-collapse:collapse;width:100%">' + s.spent.map(function (x) { var over = x.budget && x.amount > x.budget; return '<tr><td ' + td + '>' + lfEsc_(x.cat) + '</td><td ' + td + ' align="right"' + (over ? ' style="color:#c53030"' : '') + '>' + lfMoney_(x.amount) + (x.budget ? ' <span style="color:#889;font-size:12px">de ' + lfMoney_(x.budget) + '</span>' : '') + '</td></tr>'; }).join('') + '</table>';
    if (s.payments && s.payments.length) h += '<h3>Pagos de ' + lfEsc_(s.planMonth) + '</h3><table style="border-collapse:collapse;width:100%">' + s.payments.map(function (p) { var okp = p.paid + 0.005 >= p.min; return '<tr><td ' + td + '>' + lfEsc_(p.name) + '<br><span style="color:#889;font-size:12px">vence ' + lfEsc_(p.date || '') + '</span></td><td ' + td + ' align="right">' + (p.paid ? lfMoney_(p.paid) : '—') + ' / mín. ' + lfMoney_(p.min) + ' ' + (okp ? '✅' : '⏳') + '</td></tr>'; }).join('') + '</table>';
    h += '<p style="color:#889;font-size:12px">Datos de la app al ' + lfEsc_(String(s.at).slice(0, 16).replace('T', ' ')) + ' UTC.</p>';
  } else h += '<p style="color:#889">Abre la app para que el resumen incluya el estado del presupuesto y las deudas.</p>';
  if ((s && (s.pending || s.review)) || st.pending) h += '<p style="background:#fff7e6;padding:10px;border-radius:8px">🔔 Por revisar en la app: ' + ((s ? s.pending + s.review : 0) + (st.pending || 0)) + ' movimiento(s).' + (c.a ? ' <a href="' + lfEsc_(c.a) + '#/correo">Abrir la app</a>' : '') + '</p>';
  h += '<h3>Movimientos de esta semana en este correo (' + evs.length + ')</h3>';
  if (evs.length) h += '<table style="border-collapse:collapse;width:100%;font-size:13px">' + evs.map(function (e) { var inc = /transfer_in|savings_in|savings_interest|card_payment_received/.test(e.type); return '<tr><td ' + td + '>' + lfEsc_(e.date.slice(5)) + '</td><td ' + td + '>' + lfEsc_(TYPE[e.type] || e.type) + '<br><span style="color:#667">' + lfEsc_(e.type === 'card_payment' ? e.bank + ' ·' + (e.card || '') : e.type === 'bg_transfer' ? e.fromName + ' → ' + e.toName : (e.merchant || e.counterparty || e.provider || e.product || e.institution || '')) + '</span></td><td ' + td + ' align="right"' + (inc ? ' style="color:#2f855a"' : '') + '>' + (e.amount ? lfMoney_(e.amount) : '') + '</td></tr>'; }).join('') + '</table>';
  return h + '<p style="color:#99a;font-size:11px;margin-top:20px">Enviado por tu programa "Libertad Financiera" en Google Apps Script. Para dejar de recibirlo, ejecuta la función desinstalar.</p></div>';
}

// Prueba rápida: muestra el estado de la conexión.
function probar() {
  var c = lfCfg_(), st = lfRpc_(c, 'ingest_status', { p_token: c.t });
  Logger.log('Buzón: ' + st.label + ' · claves ' + (st.pk === c.p ? 'correctas' : 'DISTINTAS (genera un código nuevo)') + ' · avisos esperando en la app: ' + st.pending + ' · leyendo desde: ' + new Date(+LF_PROPS.getProperty('LF_WM') || 0));
}
